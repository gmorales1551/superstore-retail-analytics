-- Executive Level Totals (Layer 1)
CREATE OR REPLACE VIEW `executivesalesdashboard-510919.superstore_data.layer_1` AS
SELECT 
  SUM(sales) AS total_sales,
  SUM(profit) AS total_profit,
  COUNT(DISTINCT order_id) AS order_count,
  SUM(quantity) AS total_units_sold,
  SAFE_DIVIDE(SUM(sales), COUNT(DISTINCT order_id)) AS avg_order_value,
  AVG(DATE_DIFF(ship_date, order_date, DAY)) AS avg_shipping_time
FROM `executivesalesdashboard-510919.superstore_data.cleaned_totals`;
