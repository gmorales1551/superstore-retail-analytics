-- Trends and Drivers (Layer 2)
CREATE OR REPLACE VIEW `executivesalesdashboard-510919.superstore_data.layer_2` AS
SELECT
  COUNT(DISTINCT order_id) AS order_count,
  order_date,
  region,
  category,
  SUM(sales) AS total_sales,
  SUM(profit) AS total_profit
FROM `executivesalesdashboard-510919.superstore_data.cleaned_totals`
GROUP BY region,
         category,
         order_date;
